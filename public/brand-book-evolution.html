// Invalid actions leave the visit unchanged.
export function reduceSeek(c, a) {
  if (seekReason(c, a)) return c;
  const p = c.postcard;
  switch (a.type) {
    case 'SEEK_START':
      return { ...c, postcard: { ...p, room: 1 } };
    case 'SEEK_FIND':
      return { ...c, postcard: {
        ...p, found: [...p.found, a.id]
      } };
    default: return c;
  }
}
